export type StakePrecision = "whole" | "cents";
export interface ArbInput { totalStake: Decimal; oddsAmericanA: number; oddsAmericanB: number; precision: StakePrecision }
export interface ArbResult {
  impliedSum: Decimal;      // 1/Da + 1/Db, < 1 for a true arb
  stakeA: Decimal; stakeB: Decimal; totalLaid: Decimal;   // totalLaid = stakeA + stakeB <= totalStake
  payoutA: Decimal; payoutB: Decimal;                      // floored to cents
  netIfAWins: Decimal; netIfBWins: Decimal;                // payout - totalLaid
  guaranteedProfit: Decimal;                               // min(netIfAWins, netIfBWins), > 0
  returnPct: Decimal;                                      // guaranteedProfit / totalLaid * 100, 2dp ROUND_DOWN
}
export function calculateArb(input: ArbInput): ArbResult | null;

// src/domain/hedge/rankArbs.ts
export interface ArbSideQuote { bookKey: string; oddsAmerican: number }
export interface ArbSide { selection: string; point: number | null; quotes: ArbSideQuote[] }
export type ArbMarketType = "moneyline" | "spread" | "total";
export interface ArbMarket {
  eventId: string; sportKey: string; commenceTime: Date; homeTeam: string; awayTeam: string;
  marketType: ArbMarketType;
  line: number | null;   // null for moneyline; home team's signed point for spreads; the total for totals
  tieRisk: boolean;
  sideA: ArbSide;        // moneyline: away team; spread: away team at -line; total: "Over"
  sideB: ArbSide;        // moneyline: home team; spread: home team at line; total: "Under"
}
export interface BestArbPair { sideA: ArbSideQuote; sideB: ArbSideQuote; impliedSum: Decimal; tiedBooksA: string[]; tiedBooksB: string[] }

Algorithm:
- impliedSum = 1/Da + 1/Db. If impliedSum is not strictly below 1, return null. There is no epsilon and no minimum-return threshold (D-04, D-05).
- stakeAExact = clean(totalStake * (1/Da) / impliedSum) and stakeBExact = clean(totalStake - stakeAExact).
- dp is 0 for "whole" and 2 for "cents". Rounding candidates for each leg are its ROUND_DOWN and ROUND_UP values at dp, deduped when equal.
- Evaluate every (stakeA, stakeB) candidate pair. Skip a pair when either stake is at or below 0, or when stakeA + stakeB is above totalStake. The total stake is a cap (D-01, and the CONTEXT note that the arb tab's total stake is the cap). This deliberately tightens RESEARCH.md's independent-rounding sketch, which could lay more than the entered total.
- For each pair: payoutA = clean(stakeA * Da) floored to cents (ROUND_DOWN, because books pay whole cents, same as bonusBet.ts). payoutB is computed the same way. netIfAWins = payoutA - totalLaid, netIfBWins = payoutB - totalLaid, and guaranteedProfit = min of the two.
- Keep the pair with the highest guaranteedProfit. Tiebreak by lower totalLaid, then by lower stakeA. This is the Claude's-discretion rounding tiebreak: maximize worst case, then risk less money.
- If no pair survives, or the best guaranteedProfit is at or below 0, return null. Rounding erased the arb, so it must not be displayed as a guaranteed profit.
- returnPct = guaranteedProfit / totalLaid * 100 at 2dp ROUND_DOWN, based on the real amount laid. Never overstate a guaranteed number.
